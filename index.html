/** * ===================================================================== * APLIKASI PENCATATAN KEUANGAN OPERASIONAL — BACKEND (Google Apps Script) * ===================================================================== */ const SHEET_TRANSAKSI = 'Transaksi'; const SHEET_PEKERJA = 'Pekerja'; const SHEET_PENGATURAN = 'Pengaturan'; const SHEET_SETORAN = 'SetoranAwal'; const HEADER_TRANSAKSI = [ 'ID', 'Timestamp', 'Tanggal', 'JenisOmset', 'Korlap', 'Teknisi', 'Omset', 'Transfer', 'Diskon', 'Setoran', 'StatusBayar', 'Keterangan' ]; const HEADER_PEKERJA = ['Nama', 'Peran', 'KorlapInduk']; const HEADER_PENGATURAN = ['Key', 'Value']; const HEADER_SETORAN = ['NamaPekerja', 'NominalSetoranAwal']; function doGet(e) { try { ensureSheets_(); const action = (e && e.parameter && e.parameter.action) || 'getData'; let result; switch (action) { case 'getData': result = getData(e.parameter); break; case 'getWorkers': result = getWorkers(); break; case 'getSettings': result = getSettings(); break; default: result = { error: 'Unknown action: ' + action }; } return jsonResponse_(result); } catch (err) { return jsonResponse_({ error: err.message }); } } function doPost(e) { try { ensureSheets_(); const body = JSON.parse(e.postData.contents); const action = body.action; let result; switch (action) { case 'saveTransaction': result = saveTransaction(body.data); break; case 'updateTransaction': result = updateTransaction(body.data); break; case 'saveWorker': result = saveWorker(body.data); break; case 'deleteTransaction': result = deleteTransaction(body.id); break; case 'saveSetoranAwal': result = saveSetoranAwal(body.data); break; default: result = { error: 'Unknown action: ' + action }; } return jsonResponse_(result); } catch (err) { return jsonResponse_({ error: err.message }); } } function jsonResponse_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function ensureSheets_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let shTrans = ss.getSheetByName(SHEET_TRANSAKSI); if (!shTrans) { shTrans = ss.insertSheet(SHEET_TRANSAKSI); shTrans.appendRow(HEADER_TRANSAKSI); shTrans.setFrozenRows(1); } let shPekerja = ss.getSheetByName(SHEET_PEKERJA); if (!shPekerja) { shPekerja = ss.insertSheet(SHEET_PEKERJA); shPekerja.appendRow(HEADER_PEKERJA); shPekerja.setFrozenRows(1); } let shSettings = ss.getSheetByName(SHEET_PENGATURAN); if (!shSettings) { shSettings = ss.insertSheet(SHEET_PENGATURAN); shSettings.appendRow(HEADER_PENGATURAN); } let shSetoran = ss.getSheetByName(SHEET_SETORAN); if (!shSetoran) { shSetoran = ss.insertSheet(SHEET_SETORAN); shSetoran.appendRow(HEADER_SETORAN); shSetoran.setFrozenRows(1); } } function getSettings() { const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_PENGATURAN); const data = sh.getDataRange().getValues(); const settings = {}; for (let i = 1; i < data.length; i++) { if (!data[i][0]) continue; settings[data[i][0]] = Number(data[i][1]); } return settings; } function getWorkers() { const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_PEKERJA); const data = sh.getDataRange().getValues(); const workers = []; for (let i = 1; i < data.length; i++) { if (!data[i][0]) continue; workers.push({ nama: data[i][0], peran: data[i][1], korlapInduk: data[i][2] || '' }); } return workers; } function saveWorker(data) { const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_PEKERJA); sh.appendRow([data.nama, data.peran, data.korlapInduk || '']); return { success: true, workers: getWorkers() }; } function getSetoranAwalMap_() { const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_SETORAN); if (!sh) return {}; const data = sh.getDataRange().getValues(); const map = {}; for (let i = 1; i < data.length; i++) { if (data[i][0]) { map[data[i][0]] = Number(data[i][1]) || 0; } } return map; } function saveSetoranAwal(data) { const { nama, nominal } = data; const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_SETORAN); const values = sh.getDataRange().getValues(); let found = false; for (let i = 1; i < values.length; i++) { if (values[i][0] === nama) { sh.getRange(i + 1, 2).setValue(Number(nominal) || 0); found = true; break; } } if (!found) { sh.appendRow([nama, Number(nominal) || 0]); } return { success: true }; } function saveTransaction(data) { const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_TRANSAKSI); const id = 'TRX-' + new Date().getTime(); const omset = Number(data.omset) || 0; const transfer = Number(data.transfer) || 0; const diskon = Number(data.diskon) || 0; const setoran = omset - transfer - diskon; sh.appendRow([ id, new Date(), data.tanggal, data.jenisOmset, data.korlap || '', data.teknisi || '', omset, transfer, diskon, setoran, '', // Status Bayar dihapus '' // Keterangan dihapus ]); return { success: true, id: id }; } function updateTransaction(data) { const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_TRANSAKSI); const values = sh.getDataRange().getValues(); const omset = Number(data.omset) || 0; const transfer = Number(data.transfer) || 0; const diskon = Number(data.diskon) || 0; const setoran = omset - transfer - diskon; for (let i = 1; i < values.length; i++) { if (values[i][0] === data.id) { const rowIdx = i + 1; sh.getRange(rowIdx, 3).setValue(data.tanggal); sh.getRange(rowIdx, 4).setValue(data.jenisOmset); sh.getRange(rowIdx, 5).setValue(data.korlap || ''); sh.getRange(rowIdx, 6).setValue(data.teknisi || ''); sh.getRange(rowIdx, 7).setValue(omset); sh.getRange(rowIdx, 8).setValue(transfer); sh.getRange(rowIdx, 9).setValue(diskon); sh.getRange(rowIdx, 10).setValue(setoran); return { success: true }; } } return { success: false, error: 'ID transaksi tidak ditemukan' }; } function deleteTransaction(id) { const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_TRANSAKSI); const data = sh.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (data[i][0] === id) { sh.deleteRow(i + 1); return { success: true }; } } return { success: false, error: 'ID transaksi tidak ditemukan' }; } function getData(params) { params = params || {}; const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_TRANSAKSI); const values = sh.getDataRange().getValues(); const workers = getWorkers(); const setoranAwalMap = getSetoranAwalMap_(); const transactions = []; for (let i = 1; i < values.length; i++) { const row = values[i]; if (!row[0]) continue; const omset = Number(row[6]) || 0; const transfer = Number(row[7]) || 0; const diskon = Number(row[8]) || 0; const setoran = row[9] !== undefined && row[9] !== '' ? Number(row[9]) : (omset - transfer - diskon); transactions.push({ id: row[0], timestamp: row[1], tanggal: formatDate_(row[2]), jenisOmset: row[3], korlap: row[4], teknisi: row[5], omset: omset, transfer: transfer, diskon: diskon, setoran: setoran }); } let filtered = transactions.slice(); if (params.startDate) filtered = filtered.filter(function (t) { return t.tanggal >= params.startDate; }); if (params.endDate) filtered = filtered.filter(function (t) { return t.tanggal <= params.endDate; }); if (params.worker) filtered = filtered.filter(function (t) { return t.korlap === params.worker || t.teknisi === params.worker; }); // Poin 1: Total transfer masuk berasal dari customer di kolom transaksi (Transfer) const totalOmset = sumBy_(filtered, 'omset'); const totalTransferMasuk = sumBy_(filtered, 'transfer'); const korlapList = workers.filter(function (w) { return w.peran === 'Korlap'; }); const teknisiList = workers.filter(function (w) { return w.peran === 'Teknisi'; }); const rekapKorlap = korlapList.map(function (k) { return buildRekapPekerja_(k.nama, 'Korlap', filtered, setoranAwalMap[k.nama] || 0); }); const rekapTeknisi = teknisiList.map(function (t) { return buildRekapPekerja_(t.nama, 'Teknisi', filtered, setoranAwalMap[t.nama] || 0); }); return { transactions: filtered.sort(function (a, b) { return new Date(b.tanggal) - new Date(a.tanggal); }), workers: workers, summary: { totalOmset: totalOmset, totalTransferMasuk: totalTransferMasuk }, rekapKorlap: rekapKorlap, rekapTeknisi: rekapTeknisi }; } function buildRekapPekerja_(nama, peran, filtered, setoranAwal) { // Ambil transaksi milik pekerja bersangkutan const trx = filtered.filter(function (t) { if (peran === 'Korlap') return t.korlap === nama && t.jenisOmset === 'Korlap'; return t.teknisi === nama && t.jenisOmset === 'Teknisi'; }); const omset = sumBy_(trx, 'omset'); const transfer = sumBy_(trx, 'transfer'); const diskon = sumBy_(trx, 'diskon'); const sumSetoran = sumBy_(trx, 'setoran'); // Total Omset - Transfer - Diskon // Setoran Akhir = (Total Setoran Periode Ini) + (Setoran Sebelumnya) const setoranAkhir = sumSetoran + setoranAwal; return { nama: nama, peran: peran, omset: omset, transfer: transfer, diskon: diskon, sumSetoran: sumSetoran, setoranAwal: setoranAwal, setoranAkhir: setoranAkhir, statusTransfer: setoranAkhir >= 0 ? 'Teknisi / Korlap Transfer ke Owner' : 'Owner Transfer ke Teknisi / Korlap' }; } function sumBy_(arr, key) { return arr.reduce(function (s, x) { return s + (Number(x[key]) || 0); }, 0); } function formatDate_(d) { if (Object.prototype.toString.call(d) === '[object Date]') { return Utilities.formatDate(d, Session.getScriptTimeZone(), 'yyyy-MM-dd'); } return d; }
